# Pin all Ruby encodings/transcoders into the DLL; no enc/*.so installation is required.
file(GLOB ruby_encodings "${ruby_source}/enc/*.c")
list(FILTER ruby_encodings EXCLUDE REGEX "/(ascii|us_ascii|unicode|utf_8|mktable)\\.c$")
file(GLOB ruby_transcoders "${ruby_source}/enc/trans/*.c")
list(FILTER ruby_transcoders EXCLUDE REGEX "/newline\\.c$")
set_source_files_properties(${ruby_encodings} ${ruby_transcoders} PROPERTIES COMPILE_OPTIONS /w COMPILE_DEFINITIONS EXTSTATIC=1)
set(enc_init "#include <ruby.h>\nvoid ruby_init_ext(const char*, void (*)(void));\nvoid Init_encdb(void);\nvoid Init_transdb(void);\n")
set(enc_calls "void Init_enc(void) {\nInit_encdb(); rb_provide(\"enc/encdb.so\");\n")
foreach(source IN LISTS ruby_encodings ruby_transcoders)
  get_filename_component(name "${source}" NAME_WE)
  if(name STREQUAL "encdb" OR name STREQUAL "transdb")
    continue()
  endif()
  if(source MATCHES "/trans/")
    string(APPEND enc_init "void Init_trans_${name}(void);\n")
    string(APPEND enc_calls "ruby_init_ext(\"enc/trans/${name}.so\", Init_trans_${name});\n")
  else()
    string(APPEND enc_init "void Init_${name}(void);\n")
    string(APPEND enc_calls "ruby_init_ext(\"enc/${name}.so\", Init_${name});\n")
  endif()
endforeach()
string(APPEND enc_calls "Init_transdb(); rb_provide(\"enc/trans/transdb.so\");\n}\n")
file(WRITE "${CMAKE_BINARY_DIR}/encinit.c" "${enc_init}${enc_calls}")

  string(JSON fields GET "${rpg_schema}" "${name}")
  string(JSON field_count LENGTH "${fields}")
  math(EXPR field_last "${field_count}-1")
  set(field_list "")
  foreach(j RANGE ${field_last})
    string(JSON field GET "${fields}" ${j})
    string(APPEND field_list " ${field}")
  endforeach()
  string(APPEND data_schema "RGSSXP.define_data_class('${name}', %w[${field_list}])\n")
endforeach()
